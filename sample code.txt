version 3
import os
import sqlite3
import hashlib
import hmac
import logging
from contextlib import contextmanager
from typing import Optional

logger = logging.getLogger(__name__)

DATABASE_PATH = os.getenv("DATABASE_PATH", "db.sqlite3")
SECRET_KEY = os.getenv("SECRET_KEY")

if not SECRET_KEY:
    raise EnvironmentError("SECRET_KEY environment variable is not set.")


@contextmanager
def get_db_connection():
    """Context manager that ensures DB connections are always closed."""
    conn = sqlite3.connect(DATABASE_PATH)
    conn.row_factory = sqlite3.Row
    try:
        yield conn
    finally:
        conn.close()


def hash_password(password: str) -> str:
    """Hash a password using SHA-256 with a random salt."""
    salt = os.urandom(32)
    key = hashlib.pbkdf2_hmac("sha256", password.encode(), salt, 100_000)
    return salt.hex() + ":" + key.hex()


def verify_password(password: str, stored_hash: str) -> bool:
    """Constant-time password verification to prevent timing attacks."""
    salt_hex, key_hex = stored_hash.split(":")
    salt = bytes.fromhex(salt_hex)
    new_key = hashlib.pbkdf2_hmac("sha256", password.encode(), salt, 100_000)
    return hmac.compare_digest(new_key.hex(), key_hex)


def authenticate_user(username: str, password: str) -> bool:
    """
    Authenticate a user by username and password.
    Returns True if credentials are valid, False otherwise.
    """
    if not username or not password:
        raise ValueError("Username and password must not be empty.")

    try:
        with get_db_connection() as conn:
            cursor = conn.cursor()
            cursor.execute(
                "SELECT password_hash FROM users WHERE username = ?",
                (username,),
            )
            row = cursor.fetchone()

        if row is None:
            return False

        return verify_password(password, row["password_hash"])

    except sqlite3.Error as e:
        logger.error("Database error during authentication: %s", e)
        return False


def get_user_by_id(user_id: int) -> Optional[sqlite3.Row]:
    """Fetch a single user record by ID. Returns None if not found."""
    if not isinstance(user_id, int) or user_id <= 0:
        raise ValueError(f"Invalid user_id: {user_id}")

    with get_db_connection() as conn:
        cursor = conn.cursor()
        cursor.execute("SELECT id, username, email FROM users WHERE id = ?", (user_id,))
        return cursor.fetchone()


def add_unique_item(collection: list, item: object) -> list:
    """Add an item to a list only if it isn't already present."""
    if item not in collection:
        collection.append(item)
    return collection
